cmake_minimum_required(VERSION 3.20)
project(QAROS_VTK VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QARVTK_BUILD_RENDERER "Build VTK/OpenGL renderer and preview" ON)
option(QARVTK_BUILD_SENDER "Build sender against a matching local QAROS SDK" OFF)
option(QARVTK_BUILD_HOST_BRIDGE "Build VTK-free Windows host-render transport (session SDK)" OFF)
set(QAROS_SDK_ROOT "" CACHE PATH "QAROS package with include/qar_streaming.h and bin/")
include(CTest)
add_library(qarvtk_core src/math.cpp src/layout.cpp src/shutdown.cpp)
target_include_directories(qarvtk_core PUBLIC include)
if(MSVC)
  target_compile_options(qarvtk_core PRIVATE /W4)
else()
  target_compile_options(qarvtk_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(QARVTK_BUILD_HOST_BRIDGE)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Host bridge ABI 3 requires Windows x64")
  endif()
  if(NOT EXISTS "${QAROS_SDK_ROOT}/include/qar_streaming.h")
    message(FATAL_ERROR "Host bridge requires the matching session SDK header")
  endif()
  file(SHA256 "${QAROS_SDK_ROOT}/include/qar_streaming.h" host_sdk_sha)
  if(NOT host_sdk_sha STREQUAL "022c40b670f87a066664c37daeb06da525db37fffdf5810e72cff8a5b81e677d")
    message(FATAL_ERROR "Host bridge ABI 3 is pinned to the validated session-022c40b6 SDK")
  endif()
  add_library(qarvtk-host SHARED src/host_bridge.cpp src/host_gl.cpp src/gl_copy.cpp src/gl_composite.cpp)
  target_include_directories(qarvtk-host PRIVATE include "${QAROS_SDK_ROOT}/include")
  target_compile_definitions(qarvtk-host PRIVATE QARVTK_HOST_GL QARVTK_HOST_EXPORTS QAR_ENABLE_OPENGL QAR_ENABLE_DYNAMIC_LOADING NOMINMAX WIN32_LEAN_AND_MEAN
    QARVTK_HOST_SDK_HASH="${host_sdk_sha}")
  target_link_libraries(qarvtk-host PRIVATE qarvtk_core opengl32 gdi32 user32)
endif()

  find_package(Python3 COMPONENTS Interpreter)

    add_test(NAME host-runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_host_runtime.py)

    if(QARVTK_BUILD_HOST_BRIDGE)
      add_test(NAME host-abi COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_host_abi.py $<TARGET_FILE:qarvtk-host>)

    add_test(NAME plugin-process COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_runtime.py $<TARGET_FILE:qarvtk-process-fixture>)

if(QARVTK_BUILD_SENDER)
  if(APPLE)
    message(FATAL_ERROR "QAROS sender currently supports Windows/WGL and Linux/GLX only")
  endif()
  if(NOT QARVTK_BUILD_RENDERER)
    message(FATAL_ERROR "Sender requires QARVTK_BUILD_RENDERER=ON")
  endif()
  if(NOT EXISTS "${QAROS_SDK_ROOT}/include/qar_streaming.h")
    message(FATAL_ERROR "Set QAROS_SDK_ROOT to the matching QAROS runtime package")
  endif()
  file(READ "${QAROS_SDK_ROOT}/include/qar_streaming.h" sdk_header)
  if(NOT sdk_header MATCHES "qar_render_sender_frame_opengl")
    message(FATAL_ERROR "This SDK header has no OpenGL render sender")
  endif()
  if(sdk_header MATCHES "qar_runtime_onboard")
    set(QARVTK_MODERN_SDK 1)
    message(STATUS "QAROS API: Hub onboarding / render requests")
  else()
    set(QARVTK_MODERN_SDK 0)
    message(STATUS "QAROS API: legacy session invite")
  endif()
  if(sdk_header MATCHES "qar_stream_params_opengl_wgl_default")
    set(QARVTK_SPLIT_OPENGL_PARAMS 1)
  elseif(sdk_header MATCHES "qar_stream_params_opengl_default")
    set(QARVTK_SPLIT_OPENGL_PARAMS 0)
  else()
    message(FATAL_ERROR "Unsupported QAROS OpenGL context parameters")
  endif()
  file(SHA256 "${QAROS_SDK_ROOT}/include/qar_streaming.h" QARVTK_SDK_SHA256)
  configure_file(src/sdk_config.hpp.in generated/sdk_config.hpp @ONLY)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qarvtk-sdk.json" "{\"header_sha256\":\"${QARVTK_SDK_SHA256}\",\"modern_api\":${QARVTK_MODERN_SDK},\"plugin_protocol\":2}\n")
  add_executable(qarvtk-sender apps/sender.cpp)
  target_include_directories(qarvtk-sender PRIVATE "${QAROS_SDK_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(qarvtk-sender PRIVATE QAR_ENABLE_OPENGL QAR_ENABLE_DYNAMIC_LOADING)
  target_link_libraries(qarvtk-sender PRIVATE qarvtk_renderer ${CMAKE_DL_LIBS})
  if(UNIX AND NOT APPLE)
    find_package(X11 REQUIRED)
    target_link_libraries(qarvtk-sender PRIVATE X11::X11)
  elseif(WIN32)
    target_link_libraries(qarvtk-sender PRIVATE gdi32)
  endif()
  vtk_module_autoinit(TARGETS qarvtk-sender MODULES ${VTK_LIBRARIES})
endif()
