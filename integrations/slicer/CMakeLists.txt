cmake_minimum_required(VERSION 3.20)
project(QarosVTK)
find_package(Slicer REQUIRED)
include(${Slicer_USE_FILE})
set(EXTENSION_HOMEPAGE "https://github.com/HonzaHovora/QAROS_VTK")
set(EXTENSION_CATEGORY "QAROS")
set(EXTENSION_CONTRIBUTORS "QuaternAR")
set(EXTENSION_DESCRIPTION "Stream model and segmentation surfaces using a separate QAROS VTK backend.")
set(EXTENSION_ICONURL "")
set(EXTENSION_SCREENSHOTURLS "")
set(EXTENSION_DEPENDS "NA")
configure_file(QarosVTK.py ${CMAKE_CURRENT_BINARY_DIR}/QarosVTK.py COPYONLY)
configure_file(qarvtk_slicer.py ${CMAKE_CURRENT_BINARY_DIR}/qarvtk_slicer.py COPYONLY)
configure_file(../qarvtk_bridge.py ${CMAKE_CURRENT_BINARY_DIR}/qarvtk_bridge.py COPYONLY)
configure_file(../qarvtk_runtime.py ${CMAKE_CURRENT_BINARY_DIR}/qarvtk_runtime.py COPYONLY)
set(module_scripts)
foreach(script QarosVTK.py qarvtk_slicer.py qarvtk_bridge.py qarvtk_runtime.py)
  list(APPEND module_scripts ${CMAKE_CURRENT_BINARY_DIR}/${script})
endforeach()
slicerMacroBuildScriptedModule(NAME QarosVTK SCRIPTS ${module_scripts})
include(${Slicer_EXTENSION_GENERATE_CONFIG})
include(${Slicer_EXTENSION_CPACK})
